cmake_minimum_required(VERSION 3.21)
project(FreeCrawlPortableLauncher LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This compiled launcher must be built on Windows")
endif()
add_executable(FreeCrawlSEOToolPortable WIN32 launcher.cpp icon.rc)
target_compile_features(FreeCrawlSEOToolPortable PRIVATE cxx_std_17)
target_compile_definitions(FreeCrawlSEOToolPortable PRIVATE UNICODE _UNICODE)
target_link_libraries(FreeCrawlSEOToolPortable PRIVATE user32)
if(MSVC)
  target_compile_options(FreeCrawlSEOToolPortable PRIVATE /W4 /EHsc /utf-8)
  target_link_options(FreeCrawlSEOToolPortable PRIVATE /MACHINE:X64)
endif()
