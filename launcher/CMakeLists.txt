cmake_minimum_required(VERSION 3.20)
project(FreeCrawlPortableLauncher LANGUAGES CXX RC)
if(NOT DEFINED FREECRAWL_ICON OR NOT EXISTS "${FREECRAWL_ICON}")
  message(FATAL_ERROR "Set FREECRAWL_ICON to the official upstream icon.ico")
endif()
file(TO_CMAKE_PATH "${FREECRAWL_ICON}" FREECRAWL_ICON_RC)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/launcher.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" @ONLY)
add_executable(FreeCrawlPortable WIN32 FreeCrawlPortable.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
set_property(TARGET FreeCrawlPortable PROPERTY CXX_STANDARD 17)
set_property(TARGET FreeCrawlPortable PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_definitions(FreeCrawlPortable PRIVATE UNICODE _UNICODE)
target_link_libraries(FreeCrawlPortable PRIVATE user32)
if(MSVC)
  target_compile_options(FreeCrawlPortable PRIVATE /W4 /EHsc)
  target_link_options(FreeCrawlPortable PRIVATE /MANIFESTUAC:"level='asInvoker' uiAccess='false'")
endif()
